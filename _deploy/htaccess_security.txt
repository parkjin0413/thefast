# 1) HTTPS 강제 (카페24 무료 SSL 적용 후 사용)
<IfModule mod_rewrite.c>
RewriteEngine On
RewriteCond %{HTTPS} off
RewriteCond %{HTTP:X-Forwarded-Proto} !https
RewriteRule ^(.*)$ https://%{HTTP_HOST}/$1 [R=301,L]
</IfModule>

# 2) 폴더 목록 보기 차단 — 적용 후 사이트 전체가 500 오류면 이 줄만 지우세요(호스팅이 허용하지 않는 경우)
Options -Indexes

# 3) 설치 폴더 차단 — 설치가 끝나면 /install 폴더는 삭제가 가장 안전합니다. 삭제가 어렵다면 아래로 차단
<IfModule mod_rewrite.c>
RewriteRule ^install(/|$) - [F,L]
</IfModule>

# 4) 숨김 파일(.git, .env 등)·백업·문서 파일 직접 열람 차단
<IfModule mod_rewrite.c>
RewriteRule (^|/)\.(?!well-known/) - [F,L]
</IfModule>
<FilesMatch "\.(sql|bak|old|orig|swp|log|ini|sh|md|zip|tar|gz|7z)$">
    <IfModule mod_authz_core.c>
        Require all denied
    </IfModule>
    <IfModule !mod_authz_core.c>
        Order allow,deny
        Deny from all
    </IfModule>
</FilesMatch>

# 6) 보안 헤더 (이미지·CSS 같은 PHP 가 아닌 파일까지 적용)
#    테마 head.sub.php 에서도 같은 헤더를 보내므로, 서버에서 mod_headers 를 지원할 때만 추가로 사용
<IfModule mod_headers.c>
    Header always set X-Content-Type-Options "nosniff"
    Header always set X-Frame-Options "SAMEORIGIN"
    Header always set Referrer-Policy "strict-origin-when-cross-origin"
</IfModule>
